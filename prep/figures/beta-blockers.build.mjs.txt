/* One small model drives the motion and the readout. level: how much noradrenaline/adrenaline is around
   (rest, stress or exercise, high sympathetic drive). drug: none, propranolol (non-selective), metoprolol (beta-1 selective).
   Every number here is illustrative: it sets how bright and how busy things look and is never shown as a measurement. */
const AGONIST = [0.15, 0.5, 0.85];
const BLOCK = [[0, 0], [0.85, 0.85], [0.85, 0.2]]; // [beta-1, beta-2] share of receptors held by the blocker
function model(level, drug) {
  const a = AGONIST[level];
  const act = BLOCK[drug].map((b) => Math.max(0, Math.min(1, a * (1 - b * (1 - 0.5 * a))))); // competitive: more agonist partly overcomes the block
  const word = (v) => (v < 0.2 ? "baseline" : v < 0.6 ? "modest" : "strong");
  return { a, act1: act[0], act2: act[1], b1: BLOCK[drug][0], b2: BLOCK[drug][1], pka1: act[0] > 0.3, pka2: act[1] > 0.3, readout: `heart ${word(act[0])} · airway ${word(act[1])}` };
}
for (const [lv, dg, want] of [[1, 0, "heart modest · airway modest"], [1, 1, "heart baseline · airway baseline"], [1, 2, "heart baseline · airway modest"], [2, 0, "heart strong · airway strong"], [2, 2, "heart modest · airway strong"]]) {
  const r = model(lv, dg).readout;
  if (r !== want) throw new Error(`model readout ${r} != ${want}`);
}
if (!(model(2, 2).act1 < model(2, 2).act2)) throw new Error("a beta-1 blocker must dampen the heart more than the airway");
if (Math.abs(model(1, 1).act1 - model(1, 1).act2) > 1e-9) throw new Error("a non-selective blocker must treat both receptors alike");
if (!(model(2, 1).act1 > model(1, 1).act1)) throw new Error("more agonist must partly overcome a competitive blocker");

/* Noradrenaline (norepinephrine), PubChem CID 439260, C8H11NO3: the dopamine skeleton with an OH on the carbon next to the ring. */
function noradrenaline() {
  const L = 26, c = [70, 100];
  const v = (k) => [c[0] + Math.cos(((-90 + 60 * k) * Math.PI) / 180) * L, c[1] + Math.sin(((-90 + 60 * k) * Math.PI) / 180) * L];
  const dir = (deg) => [Math.cos((deg * Math.PI) / 180) * L, Math.sin((deg * Math.PI) / 180) * L];
  const add = (p, d) => [p[0] + d[0], p[1] + d[1]];
  const V = [0, 1, 2, 3, 4, 5].map(v);
  const Ca = add(V[1], dir(-30)), Cb = add(Ca, dir(30)), N = add(Cb, dir(-30));
  const OHa = add(Ca, dir(-90));
  const O3 = add(V[3], dir(90)), O4 = add(V[4], dir(150));
  const bond = (a, b) => `<path class="dr-bond" d="${pathOf([a, b])}"/>`;
  const inner = (a, b) => {
    const mx = (a[0] + b[0]) / 2, my = (a[1] + b[1]) / 2, vx = c[0] - mx, vy = c[1] - my, d = Math.hypot(vx, vy);
    const o = 4.4, s = 0.16;
    const p = (Q, R) => [Q[0] + (R[0] - Q[0]) * s + (vx / d) * o, Q[1] + (R[1] - Q[1]) * s + (vy / d) * o];
    return `<path class="dr-bond" d="${pathOf([p(a, b), p(b, a)])}"/>`;
  };
  let out = "";
  for (let k = 0; k < 6; k++) out += bond(V[k], V[(k + 1) % 6]);
  out += inner(V[0], V[1]) + inner(V[2], V[3]) + inner(V[4], V[5]);
  out += bond(V[1], Ca) + bond(Ca, Cb) + bond(Cb, N) + bond(Ca, OHa) + bond(V[3], O3) + bond(V[4], O4);
  const atom = (p, t, sub) => `<circle class="dr-atom-bg" cx="${f1(p[0])}" cy="${f1(p[1])}" r="9"/><text class="dr-atom" x="${f1(p[0])}" y="${f1(p[1] + 4)}" text-anchor="middle">${t}${sub ? `<tspan class="dr-sub" dy="3">${sub}</tspan>` : ""}</text>`;
  out += atom(O3, "OH") + atom(O4, "OH") + atom(OHa, "OH") + atom(N, "NH", "2");
  return `<g class="hot" tabindex="0" data-info="Noradrenaline (norepinephrine), 4-(2-amino-1-hydroxyethyl)benzene-1,2-diol, C8H11NO3 (PubChem CID 439260): a catechol ring with an ethanolamine side chain. It is the main transmitter of sympathetic nerves; adrenaline from the adrenal medulla acts on the same receptors. Flat skeletal formula, bond angles and lengths drawn schematically; stereochemistry is not shown. In the cleft below it is drawn as a small bright dot." aria-label="Noradrenaline structural formula">${out}</g>`
    + `<text class="dr-label dr-name" x="${f1(c[0] + 45)}" y="188" text-anchor="middle">noradrenaline</text><text class="dr-label dr-muted" x="${f1(c[0] + 45)}" y="204" text-anchor="middle">C<tspan class="dr-sub" dy="3">8</tspan><tspan dy="-3">H</tspan><tspan class="dr-sub" dy="3">11</tspan><tspan dy="-3">NO</tspan><tspan class="dr-sub" dy="3">3</tspan></text>`;
}
