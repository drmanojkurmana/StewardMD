    <script type="text/javascript">
        function setupDeleteActions() {
            $('a[data-toggle="delete"]').click(function (e) {
                var id = $(this).attr('href');
                $(id).hide();

                $(this).parent().find(id + '_visible').val('false');

                e.preventDefault();
            });
        }

        function setupColapsed() {
            $('#biometry').find('.panel-collapse').removeClass('in').addClass('collapse');
            $('#biometry').find('.panel-collapse:last').addClass('in');
        }

        function setupAgeSelectors() {
            $('#biometry input.week').change(function () {
                $(this).closest('.panel-default').find('span.week').html($(this).val());
            });

            $('#biometry input.day').change(function () {
                $(this).closest('.panel-default').find('span.day').html($(this).val());
            });
        }

        $('#add').click(function () {
            $.ajax({
                async: false,
                cache: false,
                url: '/preterm/en/ManualEntry/Create?index=' + $('#biometry').find('.panel').length
            }).success(function (partialView) {
                $('#biometry').append(partialView);
                setupDeleteActions();
                setupColapsed();
                setupAgeSelectors();
            });
        });

        setupDeleteActions();
        setupColapsed();
        setupAgeSelectors();

    </script>
